# (10 c * 9/5) + 32 = 50 F
centigrados = float (input("Introducir los grados centigrados"))
fahrenheit = (centigrados * (9/5))+32
print(centigrados, "Introducir grados centigrados convertidos a fahrenheit = ",fahrenheit)
f = float(input("Dame los grados Farenheit"))
centigrados = (f-32) * 5/9 
print(f, "Fahrenheit grads converted to celcius grad = ", centigrados)
